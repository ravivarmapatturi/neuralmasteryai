```python
import math

def sobel_edge_magnitude(img: list[list[float]]) -> list[list[float]]:
    H = len(img)
    W = len(img[0])
    out_H = H - 2
    out_W = W - 2
    mag = []
    for i in range(out_H):
        row = []
        for j in range(out_W):
            p00 = img[i][j];     p01 = img[i][j+1];     p02 = img[i][j+2]
            p10 = img[i+1][j];   p11 = img[i+1][j+1];   p12 = img[i+1][j+2]
            p20 = img[i+2][j];   p21 = img[i+2][j+1];   p22 = img[i+2][j+2]

            gx = -p00 + p02 - 2.0 * p10 + 2.0 * p12 - p20 + p22
            gy = -p00 - 2.0 * p01 - p02 + p20 + 2.0 * p21 + p22
            m = math.sqrt(gx * gx + gy * gy)
            row.append(round(float(m), 4))
        mag.append(row)
    return mag
```

Evaluating the orthogonal Sobel convolution responses Gx and Gy produces the Euclidean gradient vector magnitude at each pixel location.
