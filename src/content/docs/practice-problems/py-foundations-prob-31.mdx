```python
def validate_schema(record, schema):
    type_map = {'str': str, 'int': int, 'float': float, 'bool': bool, 'list': list, 'dict': dict}
    if not isinstance(record, dict):
        return False
    for req in schema.get('required', []):
        if req not in record:
            return False
    props = schema.get('properties', {})
    for k, v in record.items():
        if k in props:
            t = type_map.get(props[k])
            if t is not None:
                if t is int and isinstance(v, bool):
                    return False
                if not isinstance(v, t):
                    return False
    return True
```

Checking key presence and validating types against Python primitive classes provides safe schema enforcement.
