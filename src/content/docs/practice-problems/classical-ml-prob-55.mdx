```python
def roc_auc_score(y_true, y_scores):
    pairs = sorted(zip(y_scores, y_true), key=lambda x: x[0], reverse=True)
    pos_count = sum(y_true)
    neg_count = len(y_true) - pos_count
    if pos_count == 0 or neg_count == 0:
        return 0.0
    
    tp = 0
    fp = 0
    tpr_prev = 0.0
    fpr_prev = 0.0
    auc = 0.0
    
    i = 0
    while i < len(pairs):
        thresh = pairs[i][0]
        while i < len(pairs) and pairs[i][0] == thresh:
            if pairs[i][1] == 1:
                tp += 1
            else:
                fp += 1
            i += 1
        tpr = tp / pos_count
        fpr = fp / neg_count
        auc += 0.5 * (tpr + tpr_prev) * (fpr - fpr_prev)
        tpr_prev = tpr
        fpr_prev = fpr
    return round(float(auc), 4)
```

ROC-AUC equals the probability that a randomly chosen positive sample ranks higher than a randomly chosen negative sample.
