```python
def aggregate_sliding_time_windows(events, entity_id, target_metric, windows_sec):
    matched = [e for e in events if e['entity_id'] == entity_id]
    if not matched: return {}
    matched.sort(key=lambda x: x['timestamp'])
    latest = matched[-1]['timestamp']
    res = {}
    for w in windows_sec:
        vals = [e[target_metric] for e in matched if latest - w <= e['timestamp'] <= latest]
        n = len(vals)
        s = sum(vals)
        res[f'{w}s'] = {'count': n, 'sum': round(float(s), 4), 'mean': round(float(s / n if n else 0.0), 4)}
    return res
```

Filters by entity, identifies anchor timestamp, and aggregates backward window intervals.
