```python
def matrix_diagonal_summary(matrix, k):
    if not matrix or not matrix[0]:
        return {"main_diagonal": [], "k_diagonal": [], "trace": 0.0}
    rows = len(matrix)
    cols = len(matrix[0])
    main_diag = [round(float(matrix[i][i]), 4) for i in range(min(rows, cols))]
    k_diag = []
    if k >= 0:
        for i in range(rows):
            j = i + k
            if j < cols:
                k_diag.append(round(float(matrix[i][j]), 4))
    else:
        for i in range(-k, rows):
            j = i + k
            if 0 <= j < cols:
                k_diag.append(round(float(matrix[i][j]), 4))
    trace = round(float(sum(main_diag)), 4)
    return {
        "main_diagonal": main_diag,
        "k_diagonal": k_diag,
        "trace": trace
    }
```

Matrix trace is invariant under cyclic permutations and equals the sum of eigenvalues in linear algebraic representations.
