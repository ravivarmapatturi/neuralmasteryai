```python
from collections import defaultdict

def pagerank_power_iteration(n, edges, d, max_iter):
    out_degree = [0] * n
    in_edges = defaultdict(list)
    for u, v in edges:
        out_degree[u] += 1
        in_edges[v].append(u)
    p = [1.0 / n] * n
    base = (1.0 - d) / n
    for _ in range(max_iter):
        new_p = [base] * n
        for v in range(n):
            share = sum(p[u] / out_degree[u] for u in in_edges[v] if out_degree[u] > 0)
            new_p[v] += d * share
        p = new_p
    return [round(val, 4) for val in p]
```
