```python
def merge_sorted_streams(stream1, stream2):
    i, j = 0, 0
    merged = []
    while i < len(stream1) and j < len(stream2):
        if stream1[i] <= stream2[j]:
            merged.append(stream1[i])
            i += 1
        else:
            merged.append(stream2[j])
            j += 1
    if i < len(stream1):
        merged.extend(stream1[i:])
    if j < len(stream2):
        merged.extend(stream2[j:])
    return merged
```

Advancing pointers simultaneously produces an O(n + m) sorted union without full sorting recomputation.
