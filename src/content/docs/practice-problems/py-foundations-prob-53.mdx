```python
from collections import deque

def simulate_producer_consumer(events, capacity):
    queue = deque()
    log = []
    for ev in events:
        action = ev.get('type')
        if action == 'produce':
            item = ev.get('item')
            if len(queue) >= capacity:
                log.append(f"DROPPED: {item}")
            else:
                queue.append(item)
                log.append(f"PRODUCED: {item}")
        elif action == 'consume':
            if not queue:
                log.append("UNDERFLOW")
            else:
                item = queue.popleft()
                log.append(f"CONSUMED: {item}")
    return log
```

Using double-ended queues maintains strict FIFO ordering with constant-time push and pull operations.
