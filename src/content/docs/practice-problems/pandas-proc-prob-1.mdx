```python
def validate_and_coerce_schema(records, schema):
    coerced = []
    for r in records:
        row = {}
        for col, t in schema.items():
            val = r.get(col, None)
            if val is None:
                row[col] = 0 if t == 'int' else (0.0 if t == 'float' else (False if t == 'bool' else ''))
            else:
                try:
                    if t == 'int': row[col] = int(val)
                    elif t == 'float': row[col] = round(float(val), 4)
                    elif t == 'bool': row[col] = bool(val) if not isinstance(val, str) else (val.lower() in ['true', '1', 'yes'])
                    else: row[col] = str(val)
                except (ValueError, TypeError):
                    row[col] = 0 if t == 'int' else (0.0 if t == 'float' else (False if t == 'bool' else ''))
        coerced.append(row)
    return coerced
```

Attempts type parsing per column and safely falls back to standard schema defaults on conversion error.
