```python
import heapq
from collections import defaultdict

def dijkstra_shortest_paths(n, edges, src):
    adj = defaultdict(list)
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    dist = {i: float('inf') for i in range(n)}
    dist[src] = 0
    pq = [(0, src)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in adj[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
    return {k: (v if v != float('inf') else -1) for k, v in dist.items()}
```

Use a min-heap to greedily expand the vertex with the lowest tentative path distance. Relax neighboring edges whenever a shorter path is discovered.
