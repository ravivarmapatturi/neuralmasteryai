```python
import math

def cosine_similarity_matrix(queries, targets):
    q_norms = [math.sqrt(sum(x * x for x in q)) for q in queries]
    t_norms = [math.sqrt(sum(y * y for y in t)) for t in targets]
    res = []
    for i, q in enumerate(queries):
        row = []
        for j, t in enumerate(targets):
            dot = sum(a * b for a, b in zip(q, t))
            denom = q_norms[i] * t_norms[j]
            sim = dot / denom if denom > 1e-12 else 0.0
            row.append(round(float(sim), 4))
        res.append(row)
    return res
```

Precomputes L2 norms and calculates normalized pairwise dot products.
