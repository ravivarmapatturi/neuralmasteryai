```python
def sort_list(head):
    if not head or not head['next']:
        return head
    # Split at middle
    slow, fast = head, head['next']
    while fast and fast['next']:
        slow = slow['next']
        fast = fast['next']['next']
    mid = slow['next']
    slow['next'] = None
    # Recurse
    left = sort_list(head)
    right = sort_list(mid)
    # Merge
    dummy = {'val': 0, 'next': None}
    cur = dummy
    while left and right:
        if left['val'] <= right['val']:
            cur['next'] = left
            left = left['next']
        else:
            cur['next'] = right
            right = right['next']
        cur = cur['next']
    cur['next'] = left or right
    return dummy['next']
```

Top-down merge sort: split at middle with slow/fast, recurse both halves, merge sorted halves. O(n log n) time, O(log n) recursion depth.
