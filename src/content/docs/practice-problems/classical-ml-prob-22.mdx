```python
def dbscan_point_classifier(points, eps, min_samples):
    n = len(points)
    eps_sq = eps ** 2
    neighbors = [[] for _ in range(n)]
    for i in range(n):
        for j in range(n):
            d_sq = sum((x - y) ** 2 for x, y in zip(points[i], points[j]))
            if d_sq <= eps_sq:
                neighbors[i].append(j)
    
    is_core = [len(neighbors[i]) >= min_samples for i in range(n)]
    labels = []
    for i in range(n):
        if is_core[i]:
            labels.append("core")
        else:
            has_core_neighbor = any(is_core[j] for j in neighbors[i])
            if has_core_neighbor:
                labels.append("border")
            else:
                labels.append("noise")
    return labels
```

DBSCAN defines clusters as continuous dense regions of core points and reachable border points, isolating sparse outliers as noise.
