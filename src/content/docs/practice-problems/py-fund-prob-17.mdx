```python
def evaluate_boolean_query(record, query):
    def check_cond(c):
        f = c.get('field')
        op = c.get('op')
        target = c.get('val')
        if f not in record:
            return False
        val = record[f]
        if op == 'eq': return val == target
        if op == 'gt': return val > target
        if op == 'lt': return val < target
        if op == 'in': return val in target
        return False

    if 'all_of' in query and not all(check_cond(c) for c in query['all_of']):
        return False
    if 'any_of' in query and not any(check_cond(c) for c in query['any_of']):
        return False
    if 'none_of' in query and any(check_cond(c) for c in query['none_of']):
        return False
    return True
```

Composable condition predicates evaluate hierarchical boolean queries in a single inspection pass.
