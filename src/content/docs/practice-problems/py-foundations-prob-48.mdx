```python
def parse_ranges(range_str):
    if not range_str.strip():
        return []
    nums = set()
    parts = [p.strip() for p in range_str.split(',') if p.strip()]
    for p in parts:
        if '-' in p:
            sub = p.split('-')
            start, end = int(sub[0]), int(sub[1])
            for x in range(min(start, end), max(start, end) + 1):
                nums.add(x)
        else:
            nums.add(int(p))
    return sorted(nums)
```

Splitting comma-separated tokens and expanding ranges into a hash set guarantees uniqueness before a final sort.
