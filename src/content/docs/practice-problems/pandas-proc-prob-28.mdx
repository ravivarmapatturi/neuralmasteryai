```python
import math

def compute_correlation_with_mask(matrix_cols):
    cols = sorted(matrix_cols.keys())
    n = len(matrix_cols[cols[0]]) if cols else 0
    means = {c: sum(matrix_cols[c]) / n for c in cols}
    stds = {c: math.sqrt(sum((x - means[c])**2 for x in matrix_cols[c])) for c in cols}
    corr, mask = {}, {}
    for c1 in cols:
        r_row, m_row = {}, {}
        for c2 in cols:
            denom = stds[c1] * stds[c2]
            r = sum((x - means[c1]) * (y - means[c2]) for x, y in zip(matrix_cols[c1], matrix_cols[c2])) / denom if denom > 1e-12 else (1.0 if c1 == c2 else 0.0)
            r_row[c2] = round(float(r), 4)
            m_row[c2] = abs(r) >= 0.5
        corr[c1] = r_row
        mask[c1] = m_row
    return {'correlation': corr, 'significant_mask': mask}
```

Computes centered standardized cross-products for all feature pairs and applies threshold mask.
