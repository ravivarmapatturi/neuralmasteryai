```python
import re

def query_json_path(data, path):
    if not path.startswith("$."):
        return []
    tokens = []
    for part in path[2:].split('.'):
        if not part:
            continue
        m = re.match(r'^([^\[]+)(\[(\*|\d+)\])?$', part)
        if m:
            tokens.append((m.group(1), m.group(3)))
        else:
            tokens.append((part, None))
    curr = [data]
    for key, idx in tokens:
        next_curr = []
        for obj in curr:
            if isinstance(obj, dict) and key in obj:
                val = obj[key]
                if idx is None:
                    next_curr.append(val)
                elif idx == '*':
                    if isinstance(val, list):
                        next_curr.extend(val)
                else:
                    int_idx = int(idx)
                    if isinstance(val, list) and 0 <= int_idx < len(val):
                        next_curr.append(val[int_idx])
        curr = next_curr
    return curr
```

Maintaining a frontier list of current matches and expanding through wildcard arrays cleanly traverses the document tree.
