```python
import math

def vector_dot_and_angle(u, v):
    dot = sum(x * y for x, y in zip(u, v))
    norm_u = math.sqrt(sum(x ** 2 for x in u))
    norm_v = math.sqrt(sum(y ** 2 for y in v))
    if norm_u == 0 or norm_v == 0:
        angle = 0.0
    else:
        cos_val = max(-1.0, min(1.0, dot / (norm_u * norm_v)))
        angle = math.acos(cos_val)
    orthogonal = abs(dot) < 1e-7
    return {'dot': round(float(dot), 4), 'angle_radians': round(float(angle), 4), 'orthogonal': orthogonal}
```

Computes the inner product and normalized cosine angle, clamped to [-1, 1] for numerical stability.
