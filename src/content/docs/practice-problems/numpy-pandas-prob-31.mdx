```python
def batch_affine_transform(x, w, b):
    b_rows = len(x)
    d_in = len(w)
    d_out = len(w[0]) if d_in > 0 else 0
    res = []
    for row in x:
        out_row = []
        for j in range(d_out):
            s = sum(row[i] * w[i][j] for i in range(d_in)) + b[j]
            out_row.append(round(float(s), 4))
        res.append(out_row)
    return res
```
