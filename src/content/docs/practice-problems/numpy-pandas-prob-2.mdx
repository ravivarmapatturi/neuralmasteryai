Impute missing values across numerical and categorical features in tabular records using Pandas. Prerequisite: `numpy-pandas-prob-1`.

<Solution>

```python
import pandas as pd
import numpy as np

def impute_missing_values(records, num_strategy="mean", cat_strategy="mode"):
    if not records:
        return []
    df = pd.DataFrame(records)
    for col in df.columns:
        if pd.api.types.is_numeric_dtype(df[col]):
            if df[col].isna().any():
                fill_val = df[col].mean() if num_strategy == "mean" else df[col].median()
                df[col] = df[col].fillna(round(float(fill_val), 2))
        else:
            if df[col].isna().any():
                if cat_strategy == "mode":
                    mode_vals = df[col].mode()
                    fill_val = mode_vals.iloc[0] if not mode_vals.empty else "Unknown"
                else:
                    fill_val = cat_strategy
                df[col] = df[col].fillna(fill_val)
    out = []
    for r in df.to_dict(orient="records"):
        clean_r = {}
        for k, v in r.items():
            if isinstance(v, (int, float)) and np.isnan(v):
                clean_r[k] = None
            else:
                clean_r[k] = v
        out.append(clean_r)
    return out
```

Column-type-aware imputation distinguishes quantitative continuous features (suitable for mean/median replacement) from discrete categorical labels (requiring mode or sentinel token replacement).
