```python
import math

def merge_online_softmax_blocks(block_maxes, block_sums, block_weights):
    m1, m2 = block_maxes[0], block_maxes[1]
    l1, l2 = block_sums[0], block_sums[1]
    O1, O2 = block_weights[0], block_weights[1]
    m_new = max(m1, m2)
    a1 = math.exp(m1 - m_new)
    a2 = math.exp(m2 - m_new)
    l_new = a1 * l1 + a2 * l2
    O_new = [round(float((a1 * o1 + a2 * o2) / l_new), 4) for o1, o2 in zip(O1, O2)]
    return {'merged_max': round(float(m_new), 4), 'merged_sum': round(float(l_new), 4), 'merged_output': O_new}
```

Implements the online softmax correction algorithm used in tiled attention kernels.
