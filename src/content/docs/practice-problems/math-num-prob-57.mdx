```python
def simplex_pivot_step(tableau, pivot_row, pivot_col):
    m = len(tableau)
    n = len(tableau[0])
    T = [[tableau[i][j] for j in range(n)] for i in range(m)]
    pivot_val = T[pivot_row][pivot_col]
    if abs(pivot_val) < 1e-12:
        return T
    T[pivot_row] = [x / pivot_val for x in T[pivot_row]]
    for i in range(m):
        if i != pivot_row:
            factor = T[i][pivot_col]
            T[i] = [T[i][j] - factor * T[pivot_row][j] for j in range(n)]
    return [[round(float(x), 4) for x in row] for row in T]
```

Normalizes pivot row and eliminates entries in pivot column across all other rows.
