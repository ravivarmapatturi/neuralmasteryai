```python
import heapq
from collections import Counter

def huffman_encoding(text):
    if not text:
        return {}
    counts = Counter(text)
    if len(counts) == 1:
        return {text[0]: "0"}
    heap = [[weight, [char, ""]] for char, weight in counts.items()]
    heapq.heapify(heap)
    while len(heap) > 1:
        lo = heapq.heappop(heap)
        hi = heapq.heappop(heap)
        for pair in lo[1:]:
            pair[1] = '0' + pair[1]
        for pair in hi[1:]:
            pair[1] = '1' + pair[1]
        heapq.heappush(heap, [lo[0] + hi[0]] + lo[1:] + hi[1:])
    codebook = {pair[0]: pair[1] for pair in heap[0][1:]}
    return dict(sorted(codebook.items()))
```

Greedily combining lowest-frequency subtrees via a min-heap creates a prefix-free code matching information-theoretic entropy limits.
