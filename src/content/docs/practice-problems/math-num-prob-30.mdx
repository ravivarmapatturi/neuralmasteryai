```python
def conjugate_gradient_step(A, b, x, r, p):
    n = len(x)
    Ap = [sum(A[i][j] * p[j] for j in range(n)) for i in range(n)]
    pAp = sum(p[i] * Ap[i] for i in range(n))
    rr = sum(r[i] ** 2 for i in range(n))
    alpha = rr / pAp if abs(pAp) > 1e-12 else 0.0
    x_next = [x[i] + alpha * p[i] for i in range(n)]
    r_next = [r[i] - alpha * Ap[i] for i in range(n)]
    rr_next = sum(r_next[i] ** 2 for i in range(n))
    beta = rr_next / rr if rr > 1e-12 else 0.0
    p_next = [r_next[i] + beta * p[i] for i in range(n)]
    return {'x': [round(float(val), 4) for val in x_next], 'r': [round(float(val), 4) for val in r_next], 'p': [round(float(val), 4) for val in p_next]}
```

Updates solution along A-conjugate direction and computes next residual and conjugate search direction.
