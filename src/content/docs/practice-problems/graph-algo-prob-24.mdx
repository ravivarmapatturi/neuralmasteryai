```python
from collections import defaultdict

def critical_connections(n, connections):
    adj = defaultdict(list)
    for u, v in connections:
        adj[u].append(v)
        adj[v].append(u)
    tin = [-1] * n
    low = [-1] * n
    timer = 0
    bridges = []
    def dfs(u, p=-1):
        nonlocal timer
        tin[u] = low[u] = timer
        timer += 1
        for v in adj[u]:
            if v == p:
                continue
            if tin[v] != -1:
                low[u] = min(low[u], tin[v])
            else:
                dfs(v, u)
                low[u] = min(low[u], low[v])
                if low[v] > tin[u]:
                    bridges.append(sorted([u, v]))
    dfs(0)
    return sorted(bridges)
```

Perform DFS computing discovery time and low-link reachable ancestor times. Any tree edge where child cannot reach an ancestor of parent constitutes a bridge.
