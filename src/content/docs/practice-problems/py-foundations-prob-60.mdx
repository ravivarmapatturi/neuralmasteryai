```python
from collections import deque

def shortest_path_grid(grid, start, end):
    if not grid or not grid[0]:
        return -1
    R, C = len(grid), len(grid[0])
    sr, sc = start
    er, ec = end
    if grid[sr][sc] == 1 or grid[er][ec] == 1:
        return -1
    if sr == er and sc == ec:
        return 0
    queue = deque([(sr, sc, 0)])
    visited = {(sr, sc)}
    while queue:
        r, c, dist = queue.popleft()
        if r == er and c == ec:
            return dist
        for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
            nr, nc = r + dr, c + dc
            if 0 <= nr < R and 0 <= nc < C and grid[nr][nc] == 0:
                if (nr, nc) not in visited:
                    visited.add((nr, nc))
                    queue.append((nr, nc, dist + 1))
    return -1
```

BFS explores all cells at distance d before moving to distance d + 1, guaranteeing the shortest path in unweighted graphs.
