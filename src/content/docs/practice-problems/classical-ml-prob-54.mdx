```python
import math
from collections import Counter

def nmi_clustering(labels_true, labels_pred):
    n = len(labels_true)
    if n == 0:
        return 1.0
    c_true = Counter(labels_true)
    c_pred = Counter(labels_pred)
    joint = Counter(zip(labels_true, labels_pred))

    def entropy(counter):
        h = 0.0
        for cnt in counter.values():
            p = cnt / n
            if p > 0:
                h -= p * math.log(p)
        return h

    h_true = entropy(c_true)
    h_pred = entropy(c_pred)
    
    mi = 0.0
    for (t, p), cnt in joint.items():
        p_tp = cnt / n
        p_t = c_true[t] / n
        p_p = c_pred[p] / n
        if p_tp > 0:
            mi += p_tp * math.log(p_tp / (p_t * p_p))
    
    denom = math.sqrt(h_true * h_pred)
    if denom == 0.0:
        return 1.0 if h_true == h_pred else 0.0
    return round(float(mi / denom), 4)
```

NMI measures non-linear mutual dependence between partition structures, invariant to label permutations and scaled by average marginal uncertainty.
