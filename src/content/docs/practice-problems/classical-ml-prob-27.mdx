```python
from collections import Counter

def best_split_threshold(feature_values, labels):
    def gini(lbls):
        if not lbls:
            return 0.0
        cnts = Counter(lbls)
        n = len(lbls)
        return 1.0 - sum((c / n) ** 2 for c in cnts.values())

    pairs = sorted(zip(feature_values, labels), key=lambda x: x[0])
    vals = [p[0] for p in pairs]
    unique_vals = sorted(list(set(vals)))
    if len(unique_vals) <= 1:
        return unique_vals[0] if unique_vals else 0.0, 0.0

    n = len(pairs)
    best_gini = float('inf')
    best_thresh = unique_vals[0]
    for i in range(len(unique_vals) - 1):
        mid = (unique_vals[i] + unique_vals[i + 1]) / 2.0
        left = [lbl for v, lbl in pairs if v <= mid]
        right = [lbl for v, lbl in pairs if v > mid]
        weighted = (len(left) / n) * gini(left) + (len(right) / n) * gini(right)
        if weighted < best_gini:
            best_gini = weighted
            best_thresh = mid
    return round(float(best_thresh), 4), round(float(best_gini), 4)
```

Evaluating midpoints between sorted unique continuous values ensures exhaustive search over all possible discrete bipartitions.
