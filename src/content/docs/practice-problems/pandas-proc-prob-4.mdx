```python
def realign_timeseries_grid(timestamps, values, step):
    mapping = dict(zip(timestamps, values))
    curr = timestamps[0] if timestamps else 0
    end = timestamps[-1] if timestamps else 0
    ts_out, val_out = [], []
    last_val = values[0] if values else 0.0
    while curr <= end:
        ts_out.append(curr)
        if curr in mapping:
            last_val = mapping[curr]
        val_out.append(round(float(last_val), 4))
        curr += step
    return {'timestamps': ts_out, 'values': val_out}
```

Iterates on uniform time intervals and forward-fills latest observed readings.
