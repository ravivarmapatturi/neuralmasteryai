```python
import math

def rolling_pearson_correlation(series_x, series_y, window_size):
    if len(series_x) != len(series_y) or len(series_x) < window_size or window_size <= 1:
        return []
    out = []
    for i in range(window_size - 1, len(series_x)):
        wx = series_x[i - window_size + 1 : i + 1]
        wy = series_y[i - window_size + 1 : i + 1]
        mx = sum(wx) / window_size
        my = sum(wy) / window_size
        cov = sum((a - mx) * (b - my) for a, b in zip(wx, wy))
        var_x = sum((a - mx) ** 2 for a, b in zip(wx, wy))
        var_y = sum((b - my) ** 2 for a, b in zip(wx, wy))
        if var_x == 0.0 or var_y == 0.0:
            out.append(0.0)
        else:
            r = cov / math.sqrt(var_x * var_y)
            out.append(round(float(r), 4))
    return out
```

Rolling correlation captures dynamic linear coupling and regime shifts between correlated data streams.
