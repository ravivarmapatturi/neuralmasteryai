```python
from collections import deque

def distance_k(root, target_val, k):
    # Build parent pointers
    parent = {}
    target = None
    q = deque([root])
    while q:
        node = q.popleft()
        if node['val'] == target_val:
            target = node
        if node['left']:
            parent[node['left']['val']] = node
            q.append(node['left'])
        if node['right']:
            parent[node['right']['val']] = node
            q.append(node['right'])
    # BFS from target treating tree as undirected
    visited = {target['val']}
    q = deque([(target, 0)])
    result = []
    while q:
        node, dist = q.popleft()
        if dist == k:
            result.append(node['val'])
        if dist >= k:
            continue
        for nb in [node['left'], node['right']]:
            if nb and nb['val'] not in visited:
                visited.add(nb['val'])
                q.append((nb, dist+1))
        if node['val'] in parent and parent[node['val']]['val'] not in visited:
            p = parent[node['val']]
            visited.add(p['val'])
            q.append((p, dist+1))
    return sorted(result)
```

Two-pass: BFS to find parent pointers + target node; then BFS from target over the undirected graph (children + parent). O(n) time.
