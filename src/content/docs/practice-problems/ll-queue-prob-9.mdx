```python
def min_stack_ops(operations):
    stack = []
    min_stack = []
    out = []
    for op in operations:
        action = op.get('op')
        if action == 'push':
            v = op.get('val')
            stack.append(v)
            min_stack.append(min(v, min_stack[-1]) if min_stack else v)
        elif action == 'pop':
            if stack:
                stack.pop()
                min_stack.pop()
        elif action == 'top':
            out.append(stack[-1] if stack else None)
        elif action == 'get_min':
            out.append(min_stack[-1] if min_stack else None)
    return out
```

Maintaining parallel value and prefix-minimum stacks guarantees constant O(1) query time for get_min.
