```python
def one_hot_encode(categories, vocabulary, handle_unknown="ignore"):
    vocab_map = {cat: i for i, cat in enumerate(vocabulary)}
    v_len = len(vocabulary)
    res = []
    for cat in categories:
        if cat in vocab_map:
            vec = [0] * v_len
            vec[vocab_map[cat]] = 1
            res.append(vec)
        else:
            if handle_unknown == "error":
                raise ValueError(f"Unknown category: {cat}")
            elif handle_unknown == "ignore":
                res.append([0] * v_len)
            else:
                raise ValueError(f"Unsupported handle_unknown mode: {handle_unknown}")
    return res
```

One-hot encoding removes false ordinal assumptions from categorical variables, creating orthogonal feature spaces for linear models and trees.
