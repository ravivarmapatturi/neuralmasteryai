```python
from collections import defaultdict

def build_markov_transition_matrix(sequences):
    counts = defaultdict(lambda: defaultdict(int))
    states = set()
    for seq in sequences:
        for i in range(len(seq) - 1):
            counts[seq[i]][seq[i+1]] += 1
            states.add(seq[i])
            states.add(seq[i+1])
    sorted_states = sorted(list(states))
    mat = {}
    for sf in sorted_states:
        tot = sum(counts[sf].values())
        mat[sf] = {st: round(float(counts[sf][st] / tot if tot else 0.0), 4) for st in sorted_states}
    return mat
```

Tallies bigram state transitions and computes row-wise conditional probabilities.
