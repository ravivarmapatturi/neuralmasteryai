```python
from collections import deque

def max_network_flow(n, source, sink, capacities):
    cap = [row[:] for row in capacities]
    max_flow = 0
    while True:
        parent = [-1] * n
        q = deque([source])
        parent[source] = source
        while q and parent[sink] == -1:
            u = q.popleft()
            for v in range(n):
                if parent[v] == -1 and cap[u][v] > 0:
                    parent[v] = u
                    q.append(v)
        if parent[sink] == -1:
            break
        path_flow = float('inf')
        curr = sink
        while curr != source:
            prev = parent[curr]
            path_flow = min(path_flow, cap[prev][curr])
            curr = prev
        curr = sink
        while curr != source:
            prev = parent[curr]
            cap[prev][curr] -= path_flow
            cap[curr][prev] += path_flow
            curr = prev
        max_flow += path_flow
    return max_flow
```

Repeatedly find augmenting paths in the residual graph via BFS, augmenting along the bottleneck edge and maintaining backward flow channels until exhaustion.
