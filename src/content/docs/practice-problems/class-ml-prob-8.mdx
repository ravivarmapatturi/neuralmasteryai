```python
import math

def pairwise_distances(X, metric):
    n = len(X)
    dist = [[0.0] * n for _ in range(n)]
    for i in range(n):
        for j in range(n):
            if metric == 'euclidean':
                d = math.sqrt(sum((a - b) ** 2 for a, b in zip(X[i], X[j])))
            else:
                d = sum(abs(a - b) for a, b in zip(X[i], X[j]))
            dist[i][j] = round(d, 4)
    return dist
```
