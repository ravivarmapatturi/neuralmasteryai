```python
def connected_components_count(n, edges):
    parent = list(range(n))
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    def union(x, y):
        rx, ry = find(x), find(y)
        if rx != ry:
            parent[rx] = ry
            return True
        return False
    components = n
    for u, v in edges:
        if union(u, v):
            components -= 1
    return components
```

Path compression flattens tree depth during find operations, guaranteeing near-O(1) amortized union steps.
