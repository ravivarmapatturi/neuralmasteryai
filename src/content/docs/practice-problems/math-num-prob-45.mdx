```python
def importance_sampling(f_evals, p_densities, q_densities):
    weights = [p / q for p, q in zip(p_densities, q_densities)]
    sum_w = sum(weights)
    if sum_w < 1e-12:
        return {'estimate': 0.0, 'effective_sample_size': 0.0}
    estimate = sum(w * f for w, f in zip(weights, f_evals)) / sum_w
    sum_w_sq = sum(w ** 2 for w in weights)
    ess = (sum_w ** 2) / sum_w_sq if sum_w_sq > 1e-12 else 0.0
    return {'estimate': round(float(estimate), 4), 'effective_sample_size': round(float(ess), 4)}
```
