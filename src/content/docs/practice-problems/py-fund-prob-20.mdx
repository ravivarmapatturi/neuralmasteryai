```python
def safe_cast_dict(records, types):
    out = []
    for r in records:
        new_r = dict(r)
        for field, target in types.items():
            if field in new_r:
                val = new_r[field]
                try:
                    if target == 'int': new_r[field] = int(val)
                    elif target == 'float': new_r[field] = float(val)
                    elif target == 'str': new_r[field] = str(val)
                    elif target == 'bool':
                        new_r[field] = val.lower() in ('true', '1', 'yes') if isinstance(val, str) else bool(val)
                except (ValueError, TypeError):
                    new_r[field] = None
        out.append(new_r)
    return out
```

Defensive type casting handles corrupted telemetry rows gracefully without terminating batch jobs.
