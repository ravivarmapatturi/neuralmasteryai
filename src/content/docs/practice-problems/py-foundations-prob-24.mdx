```python
def flatten_dict(nested, sep="."):
    flat = {}
    def recurse(curr, prefix):
        for k, v in curr.items():
            new_key = f"{prefix}{sep}{k}" if prefix else str(k)
            if isinstance(v, dict) and v:
                recurse(v, new_key)
            else:
                flat[new_key] = v
    recurse(nested, "")
    return flat
```

A depth-first recursive walk records concatenated string paths for leaf values while visiting every sub-branch.
