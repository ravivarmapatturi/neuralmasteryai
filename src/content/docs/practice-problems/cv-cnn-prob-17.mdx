```python
import math

def generate_anchor_boxes(feature_map_h: int, feature_map_w: int, stride: int, base_size: int, aspect_ratios: list[float], scales: list[float]) -> list[list[float]]:
    boxes = []
    for r in range(feature_map_h):
        cy = (r + 0.5) * stride
        for c in range(feature_map_w):
            cx = (c + 0.5) * stride
            for s in scales:
                for ar in aspect_ratios:
                    w = base_size * s * math.sqrt(ar)
                    h = base_size * s / math.sqrt(ar)
                    x1 = round(cx - w / 2.0, 2)
                    y1 = round(cy - h / 2.0, 2)
                    x2 = round(cx + w / 2.0, 2)
                    y2 = round(cy + h / 2.0, 2)
                    boxes.append([x1, y1, x2, y2])
    return boxes
```

Iterating across grid coordinates and combining scales with aspect ratios generates the canonical set of anchor proposals.
