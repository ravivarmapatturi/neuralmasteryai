```python
import math

def lstm_cell_forward(x: list[list[float]], h_prev: list[list[float]], c_prev: list[list[float]], W_x: list[list[float]], W_h: list[list[float]], b: list[float]) -> dict:
    N = len(x)
    D_in = len(W_x)
    H = len(h_prev[0])

    def sigmoid(v):
        return 1.0 / (1.0 + math.exp(-max(min(v, 30.0), -30.0)))

    next_h = []
    next_c = []
    for n in range(N):
        raw_gates = []
        for j in range(4 * H):
            v = sum(x[n][k] * W_x[k][j] for k in range(D_in)) + sum(h_prev[n][k] * W_h[k][j] for k in range(H)) + b[j]
            raw_gates.append(v)

        i_gate = [sigmoid(raw_gates[j]) for j in range(0, H)]
        f_gate = [sigmoid(raw_gates[j]) for j in range(H, 2 * H)]
        g_gate = [math.tanh(raw_gates[j]) for j in range(2 * H, 3 * H)]
        o_gate = [sigmoid(raw_gates[j]) for j in range(3 * H, 4 * H)]

        c_row = []
        h_row = []
        for j in range(H):
            c_val = f_gate[j] * c_prev[n][j] + i_gate[j] * g_gate[j]
            h_val = o_gate[j] * math.tanh(c_val)
            c_row.append(round(float(c_val), 4))
            h_row.append(round(float(h_val), 4))
        next_c.append(c_row)
        next_h.append(h_row)

    return {"h_next": next_h, "c_next": next_c}
```

LSTMs use additive linear cell updates modulated by sigmoid gates to control memory retention and update.
