```python
def pagerank_step(adj_matrix, p, damping=0.85):
    n = len(p)
    col_sums = [sum(adj_matrix[i][j] for i in range(n)) for j in range(n)]
    M = [[0.0] * n for _ in range(n)]
    for j in range(n):
        if col_sums[j] > 0:
            for i in range(n):
                M[i][j] = adj_matrix[i][j] / col_sums[j]
        else:
            for i in range(n):
                M[i][j] = 1.0 / n
    Mp = [sum(M[i][j] * p[j] for j in range(n)) for i in range(n)]
    base = (1.0 - damping) / n
    p_next = [damping * Mp[i] + base for i in range(n)]
    return [round(float(v), 4) for v in p_next]
```
