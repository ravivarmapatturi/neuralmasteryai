```python
def conv2d_discrete(image, kernel, stride=1, padding=0):
    h = len(image)
    w = len(image[0])
    kh = len(kernel)
    kw = len(kernel[0])
    if padding > 0:
        padded = [[0.0] * (w + 2 * padding) for _ in range(padding)]
        for row in image:
            padded.append([0.0] * padding + list(row) + [0.0] * padding)
        for _ in range(padding):
            padded.append([0.0] * (w + 2 * padding))
    else:
        padded = image
    out_h = (len(padded) - kh) // stride + 1
    out_w = (len(padded[0]) - kw) // stride + 1
    result = []
    for i in range(out_h):
        row = []
        for j in range(out_w):
            r_start = i * stride
            c_start = j * stride
            val = sum(padded[r_start + p][c_start + q] * kernel[p][q] for p in range(kh) for q in range(kw))
            row.append(round(float(val), 4))
        result.append(row)
    return result
```

Pads outer border of image matrix and computes spatial dot products across all stride offsets.
