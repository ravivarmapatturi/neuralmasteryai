```python
def reduce_chunked_aggregations(chunks, agg_field):
    cnt = 0
    s = 0.0
    min_v, max_v = float('inf'), float('-inf')
    for ch in chunks:
        for r in ch:
            if agg_field in r and r[agg_field] is not None:
                v = float(r[agg_field])
                cnt += 1
                s += v
                if v < min_v: min_v = v
                if v > max_v: max_v = v
    if cnt == 0:
        return {'count': 0, 'sum': 0.0, 'mean': 0.0, 'min': 0.0, 'max': 0.0}
    return {'count': cnt, 'sum': round(float(s), 4), 'mean': round(float(s / cnt), 4), 'min': round(float(min_v), 4), 'max': round(float(max_v), 4)}
```

Updates running totals across batch chunks without materializing the full dataset.
