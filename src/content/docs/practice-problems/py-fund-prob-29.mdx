```python
from collections import defaultdict, OrderedDict

def lfu_cache_simulator(operations, capacity):
    if capacity <= 0:
        return [-1 for op in operations if op.get('op') == 'get']
    key_val, key_freq = {}, {}
    freq_keys = defaultdict(OrderedDict)
    min_freq = 0
    out = []
    for op in operations:
        action = op.get('op')
        key = op.get('key')
        if action == 'get':
            if key in key_val:
                val = key_val[key]
                f = key_freq[key]
                key_freq[key] = f + 1
                del freq_keys[f][key]
                freq_keys[f + 1][key] = True
                if f == min_freq and not freq_keys[f]:
                    min_freq += 1
                out.append(val)
            else:
                out.append(-1)
        elif action == 'put':
            val = op.get('val')
            if key in key_val:
                key_val[key] = val
                f = key_freq[key]
                key_freq[key] = f + 1
                del freq_keys[f][key]
                freq_keys[f + 1][key] = True
                if f == min_freq and not freq_keys[f]:
                    min_freq += 1
            else:
                if len(key_val) >= capacity:
                    evict_k, _ = freq_keys[min_freq].popitem(last=False)
                    del key_val[evict_k]
                    del key_freq[evict_k]
                key_val[key] = val
                key_freq[key] = 1
                freq_keys[1][key] = True
                min_freq = 1
    return out
```

Combining key-frequency maps with frequency-indexed OrderedDicts achieves O(1) time complexity for both get and put operations.
