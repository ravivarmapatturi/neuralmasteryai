```python
from collections import defaultdict

def target_encoding_smooth(categories, targets, m_weight):
    global_mean = sum(targets) / len(targets) if targets else 0.0
    cat_counts = defaultdict(int)
    cat_sums = defaultdict(float)
    for c, y in zip(categories, targets):
        cat_counts[c] += 1
        cat_sums[c] += y
    encoded = []
    for c in categories:
        cnt = cat_counts[c]
        s = cat_sums[c]
        val = (s + m_weight * global_mean) / (cnt + m_weight)
        encoded.append(round(float(val), 4))
    return encoded
```

M-estimate smoothing shrinks category empirical means toward the global baseline according to category sample size, preventing target leakage on sparse levels.
