```python
import math

def gmm_em_step_1d(X, weights, means, variances):
    K, N = len(weights), len(X)
    def pdf(x, mu, var):
        return math.exp(-((x - mu) ** 2) / (2 * var)) / math.sqrt(2 * math.pi * var)
    gamma = []
    for x in X:
        likes = [weights[k] * pdf(x, means[k], variances[k]) for k in range(K)]
        total = sum(likes)
        gamma.append([l / total for l in likes])
    new_w, new_m, new_v = [], [], []
    for k in range(K):
        Nk = sum(gamma[i][k] for i in range(N))
        new_w.append(round(Nk / N, 4))
        mu_k = sum(gamma[i][k] * X[i] for i in range(N)) / Nk
        new_m.append(round(mu_k, 4))
        var_k = sum(gamma[i][k] * ((X[i] - mu_k) ** 2) for i in range(N)) / Nk
        new_v.append(round(var_k, 4))
    return {'weights': new_w, 'means': new_m, 'variances': new_v}
```

Compute posterior responsibility weights for each observation and component in the E-step, then calculate soft-weighted averages to update component parameters in the M-step.
