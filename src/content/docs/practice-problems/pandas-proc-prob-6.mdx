```python
from collections import defaultdict

def smooth_target_encoding(categories, targets, m_prior):
    gm = sum(targets) / len(targets) if targets else 0.0
    cat_map = defaultdict(list)
    for c, t in zip(categories, targets):
        cat_map[c].append(t)
    enc = {}
    for c, v in cat_map.items():
        n = len(v)
        enc[c] = round(float((n * (sum(v) / n) + m_prior * gm) / (n + m_prior)), 4)
    return {'global_mean': round(float(gm), 4), 'encoding_map': enc, 'encoded_series': [enc[c] for c in categories]}
```

Blends category sample mean with global prior mean weighted by sample size and m_prior parameter.
