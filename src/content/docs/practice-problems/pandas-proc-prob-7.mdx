```python
from collections import defaultdict

def pivot_table_with_margins(records, index_col, col_col, val_col):
    r_keys = sorted(list(set(r[index_col] for r in records)))
    c_keys = sorted(list(set(r[col_col] for r in records)))
    sums = defaultdict(float)
    for r in records:
        sums[(r[index_col], r[col_col])] += float(r[val_col])
    table = {}
    c_totals = defaultdict(float)
    for rk in r_keys:
        row_dict = {}
        r_tot = 0.0
        for ck in c_keys:
            val = sums[(rk, ck)]
            row_dict[ck] = round(float(val), 4)
            r_tot += val
            c_totals[ck] += val
        row_dict['ALL'] = round(float(r_tot), 4)
        table[rk] = row_dict
    all_row = {ck: round(float(c_totals[ck]), 4) for ck in c_keys}
    all_row['ALL'] = round(float(sum(c_totals.values())), 4)
    table['ALL'] = all_row
    return table
```

Pivots records into 2D dict and appends row, column, and grand totals.
