```python
def reconcile_dataframes(source_df, target_df, primary_key):
    src = {r[primary_key]: r for r in source_df}
    tgt = {r[primary_key]: r for r in target_df}
    added = sorted([k for k in tgt if k not in src])
    removed = sorted([k for k in src if k not in tgt])
    discrepancies = []
    for k in sorted(src.keys()):
        if k in tgt:
            cols = sorted(set(src[k].keys()) | set(tgt[k].keys()))
            for c in cols:
                if c != primary_key and src[k].get(c) != tgt[k].get(c):
                    discrepancies.append({'key': k, 'field': c, 'source': src[k].get(c), 'target': tgt[k].get(c)})
    return {'added_keys': added, 'removed_keys': removed, 'discrepancies': discrepancies}
```

Builds key lookup dictionaries and conducts two-way differential audit of keys and attributes.
