```python
def split_multi_head_projections(hidden_states, num_heads, head_dim):
    B = len(hidden_states)
    S = len(hidden_states[0]) if B > 0 else 0
    out = []
    for b in range(B):
        heads = [[] for _ in range(num_heads)]
        for s in range(S):
            token = hidden_states[b][s]
            for h in range(num_heads):
                heads[h].append(token[h * head_dim:(h + 1) * head_dim])
        out.append(heads)
    return out
```

Iterates through tokens, slices head segments, and organizes by batch and head.
