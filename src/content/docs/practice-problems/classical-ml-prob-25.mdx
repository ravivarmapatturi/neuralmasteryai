```python
import math
from collections import Counter

def entropy_info_gain(parent_labels, left_labels, right_labels):
    def entropy(lbls):
        if not lbls:
            return 0.0
        counts = Counter(lbls)
        n = len(lbls)
        h = -sum((c / n) * math.log2(c / n) for c in counts.values() if c > 0)
        return h

    h_parent = entropy(parent_labels)
    n_total = len(parent_labels)
    if n_total == 0:
        return 0.0
    weighted_children = (len(left_labels) / n_total) * entropy(left_labels) + (len(right_labels) / n_total) * entropy(right_labels)
    gain = h_parent - weighted_children
    return round(float(gain), 4)
```

Information gain quantifies uncertainty reduction about target classes upon splitting on a feature condition.
