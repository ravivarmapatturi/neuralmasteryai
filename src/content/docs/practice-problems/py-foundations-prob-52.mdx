```python
def compute_word_diff(tokens_a, tokens_b):
    m, n = len(tokens_a), len(tokens_b)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if tokens_a[i - 1] == tokens_b[j - 1]:
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
    diff = []
    i, j = m, n
    while i > 0 or j > 0:
        if i > 0 and j > 0 and tokens_a[i - 1] == tokens_b[j - 1]:
            diff.append({'op': 'keep', 'word': tokens_a[i - 1]})
            i -= 1
            j -= 1
        elif j > 0 and (i == 0 or dp[i][j - 1] >= dp[i - 1][j]):
            diff.append({'op': 'add', 'word': tokens_b[j - 1]})
            j -= 1
        elif i > 0 and (j == 0 or dp[i][j - 1] < dp[i - 1][j]):
            diff.append({'op': 'remove', 'word': tokens_a[i - 1]})
            i -= 1
    diff.reverse()
    return diff
```
