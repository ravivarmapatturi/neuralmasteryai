```python
def simulate_ttl_cache(operations, default_ttl):
    cache = {}
    out = []
    for op in operations:
        name = op.get('op')
        k = op.get('key')
        t = op.get('time', 0)
        if name == 'set':
            v = op.get('val')
            ttl = op.get('ttl', default_ttl)
            cache[k] = (v, t + ttl)
        elif name == 'get':
            if k in cache:
                val, expiry = cache[k]
                if t < expiry:
                    out.append(val)
                else:
                    del cache[k]
                    out.append(None)
            else:
                out.append(None)
    return out
```

Recording absolute expiration timestamps alongside values enables O(1) expiration checks on retrieval.
