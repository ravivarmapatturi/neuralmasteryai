```python
import math

def decode_and_cross_entropy(y_pred, y_true_onehot):
    n = len(y_pred)
    decoded = []
    loss = 0.0
    for p_row, t_row in zip(y_pred, y_true_onehot):
        t_idx = t_row.index(max(t_row))
        p_idx = p_row.index(max(p_row))
        decoded.append(p_idx)
        p = max(1e-12, min(1.0 - 1e-12, p_row[t_idx]))
        loss -= math.log(p)
    return {'decoded_labels': decoded, 'cross_entropy': round(float(loss / n), 4)}
```

Picks argmax predicted indices and averages negative log-probabilities of the target classes.
