```python
def is_orthogonal_matrix(A, tol=1e-5):
    n = len(A)
    for i in range(n):
        for j in range(n):
            dot = sum(A[k][i] * A[k][j] for k in range(n))
            expected = 1.0 if i == j else 0.0
            if abs(dot - expected) > tol:
                return False
    return True
```

Verifies that all columns form an orthonormal system via pairwise dot products.
