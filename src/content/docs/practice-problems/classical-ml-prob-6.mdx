```python
def wcss_inertia(points, centroids, assignments):
    total = 0.0
    for p, c_idx in zip(points, assignments):
        c = centroids[c_idx]
        total += sum((pi - ci) ** 2 for pi, ci in zip(p, c))
    return round(total, 6)
```
