```python
def min_cost_connect(n, connections):
    connections.sort(key=lambda x: x[2])
    parent = list(range(n + 1))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        parent[px] = py
        return True

    cost = edges_used = 0
    for u, v, w in connections:
        if union(u, v):
            cost += w
            edges_used += 1
            if edges_used == n - 1:
                return cost
    return -1
```

Kruskal's: sort edges by weight, add the cheapest edge that connects two different components (using Union-Find to detect cycles). Stop when n-1 edges are used. O(E log E) time.
