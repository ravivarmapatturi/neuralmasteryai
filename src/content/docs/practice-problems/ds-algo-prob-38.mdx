```python
from collections import deque

def walls_and_gates(rooms):
    import copy
    INF = 2147483647
    grid = copy.deepcopy(rooms)
    rows, cols = len(grid), len(grid[0])
    q = deque()
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 0:
                q.append((r, c))
    while q:
        r, c = q.popleft()
        for dr, dc in [(0,1),(0,-1),(1,0),(-1,0)]:
            nr, nc = r+dr, c+dc
            if 0<=nr<rows and 0<=nc<cols and grid[nr][nc]==INF:
                grid[nr][nc] = grid[r][c] + 1
                q.append((nr,nc))
    return grid
```

Initialize the BFS queue with all gates (value 0). Expand outward, overwriting INF cells with their distance. Multi-source BFS gives optimal shortest distances to all gates simultaneously.
