```python
import math

def bilinear_interpolate(grid, x, y):
    H, W = len(grid), len(grid[0])
    x0 = int(math.floor(x))
    x1 = min(x0 + 1, W - 1)
    y0 = int(math.floor(y))
    y1 = min(y0 + 1, H - 1)
    wx, wy = x - x0, y - y0
    v00, v01 = grid[y0][x0], grid[y0][x1]
    v10, v11 = grid[y1][x0], grid[y1][x1]
    val = (1.0 - wx) * (1.0 - wy) * v00 + wx * (1.0 - wy) * v01 + (1.0 - wx) * wy * v10 + wx * wy * v11
    return round(float(val), 4)
```

Applies linear interpolation sequentially along x and y dimensions.
