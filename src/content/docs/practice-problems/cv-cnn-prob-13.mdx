```python
def depthwise_conv2d(x: list[list[list[float]]], kernels: list[list[list[float]]], stride: int = 1) -> list[list[list[float]]]:
    C = len(x)
    H = len(x[0])
    W = len(x[0][0])
    kH = len(kernels[0])
    kW = len(kernels[0][0])

    out_H = (H - kH) // stride + 1
    out_W = (W - kW) // stride + 1

    out = []
    for c in range(C):
        c_mat = []
        for i in range(out_H):
            row = []
            for j in range(out_W):
                h_start = i * stride
                w_start = j * stride
                val = sum(x[c][h_start + kh][w_start + kw] * kernels[c][kh][kw] for kh in range(kH) for kw in range(kW))
                row.append(round(float(val), 4))
            c_mat.append(row)
        out.append(c_mat)
    return out
```

Depthwise convolution applies an independent 2D spatial convolution per channel without cross-channel communication.
