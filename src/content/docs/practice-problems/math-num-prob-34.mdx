```python
import math

def qr_decomposition(A):
    m = len(A)
    n = len(A[0])
    cols = [[A[i][j] for i in range(m)] for j in range(n)]
    Q_cols = []
    R = [[0.0] * n for _ in range(n)]
    for j in range(n):
        v = list(cols[j])
        for i in range(j):
            q_i = Q_cols[i]
            r_ij = sum(q_i[k] * cols[j][k] for k in range(m))
            R[i][j] = r_ij
            v = [v[k] - r_ij * q_i[k] for k in range(m)]
        r_jj = math.sqrt(sum(x ** 2 for x in v))
        R[j][j] = r_jj
        if r_jj > 1e-9:
            q_j = [x / r_jj for x in v]
        else:
            q_j = [0.0] * m
        Q_cols.append(q_j)
    Q = [[round(float(Q_cols[j][i]), 4) for j in range(n)] for i in range(m)]
    R_res = [[round(float(R[i][j]), 4) for j in range(n)] for i in range(n)]
    return {'Q': Q, 'R': R_res}
```

Executes modified Gram-Schmidt across matrix columns to populate Q and R.
