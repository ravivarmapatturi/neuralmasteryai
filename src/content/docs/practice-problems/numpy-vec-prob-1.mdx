```python
def conv2d_valid(image, kernel):
    H = len(image)
    W = len(image[0]) if H > 0 else 0
    kH = len(kernel)
    kW = len(kernel[0]) if kH > 0 else 0
    out = []
    for i in range(H - kH + 1):
        row = []
        for j in range(W - kW + 1):
            val = sum(image[i + ki][j + kj] * kernel[ki][kj] for ki in range(kH) for kj in range(kW))
            row.append(round(float(val), 4))
        out.append(row)
    return out
```
