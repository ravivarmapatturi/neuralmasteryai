```python
def custom_json_serializer(data):
    if data is None: return "null"
    if isinstance(data, bool): return "true" if data else "false"
    if isinstance(data, (int, float)): return str(data)
    if isinstance(data, str):
        escaped = data.replace('\\', '\\\\').replace('"', '\\"').replace('\n', '\\n').replace('\t', '\\t')
        return f'"{escaped}"'
    if isinstance(data, list):
        return f'[{", ".join(custom_json_serializer(x) for x in data)}]'
    if isinstance(data, dict):
        return f'{{{", ".join(f"\"{k}\": {custom_json_serializer(v)}" for k, v in data.items())}}}'
    return "null"
```

Checking booleans before integers and recursively formatting nested composites generates valid JSON text.
