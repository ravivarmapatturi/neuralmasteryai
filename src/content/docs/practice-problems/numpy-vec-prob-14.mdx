```python
import math

def gram_schmidt_orthonormal(vectors):
    ortho = []
    for v in vectors:
        proj_sum = [0.0] * len(v)
        for u in ortho:
            dot_vu = sum(a * b for a, b in zip(v, u))
            dot_uu = sum(b * b for b in u)
            if dot_uu > 1e-12:
                c = dot_vu / dot_uu
                proj_sum = [p + c * b for p, b in zip(proj_sum, u)]
        u_new = [a - b for a, b in zip(v, proj_sum)]
        norm_u = math.sqrt(sum(a * a for a in u_new))
        if norm_u > 1e-10:
            ortho.append([round(float(a / norm_u), 4) for a in u_new])
        else:
            ortho.append([0.0] * len(v))
    return ortho
```

Progressively subtracts projections onto established orthonormal vectors and normalizes residuals.
