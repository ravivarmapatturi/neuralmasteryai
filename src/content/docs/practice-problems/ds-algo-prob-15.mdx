```python
def queue_using_stacks(operations):
    class MyQueue:
        def __init__(self):
            self.in_stack = []
            self.out_stack = []
        def push(self, x):
            self.in_stack.append(x)
        def pop(self):
            self._transfer()
            return self.out_stack.pop()
        def peek(self):
            self._transfer()
            return self.out_stack[-1]
        def empty(self):
            return not self.in_stack and not self.out_stack
        def _transfer(self):
            if not self.out_stack:
                while self.in_stack:
                    self.out_stack.append(self.in_stack.pop())

    q = MyQueue()
    results = []
    for op in operations:
        name = op[0]
        if name == 'push':
            q.push(op[1])
            results.append(None)
        elif name == 'pop':
            results.append(q.pop())
        elif name == 'peek':
            results.append(q.peek())
        elif name == 'empty':
            results.append(q.empty())
    return results
```

Lazy transfer: only move from in_stack to out_stack when out_stack is empty. Each element crosses once: amortized O(1) per operation.
