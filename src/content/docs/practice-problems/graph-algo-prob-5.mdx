```python
from collections import deque

def is_bipartite_bfs(n, graph):
    color = [-1] * n
    for start in range(n):
        if color[start] == -1:
            color[start] = 0
            q = deque([start])
            while q:
                u = q.popleft()
                for v in graph[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        return False
    return True
```

Perform BFS 2-coloring across all connected components. When visiting an uncolored neighbor, assign the opposite color; if an already-colored neighbor has the same color, an odd cycle is detected.
