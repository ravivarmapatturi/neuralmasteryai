```python
def trie_ops(operations):
    class TrieNode:
        def __init__(self):
            self.children = {}
            self.is_end = False

    root = TrieNode()

    def insert(word):
        cur = root
        for c in word:
            if c not in cur.children:
                cur.children[c] = TrieNode()
            cur = cur.children[c]
        cur.is_end = True

    def search(word):
        cur = root
        for c in word:
            if c not in cur.children:
                return False
            cur = cur.children[c]
        return cur.is_end

    def starts_with(prefix):
        cur = root
        for c in prefix:
            if c not in cur.children:
                return False
            cur = cur.children[c]
        return True

    results = []
    for op in operations:
        name = op[0]
        if name == 'insert':
            insert(op[1])
            results.append(None)
        elif name == 'search':
            results.append(search(op[1]))
        elif name == 'starts_with':
            results.append(starts_with(op[1]))
    return results
```

Each node holds a dict of children and an `is_end` flag. insert/search/starts_with walk the trie character by character. O(L) per operation where L is the word/prefix length.
