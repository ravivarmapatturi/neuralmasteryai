```python
from collections import defaultdict

def find_all_ladders(beginWord, endWord, wordList):
    word_set = set(wordList)
    if endWord not in word_set:
        return []
    adj = defaultdict(list)
    layer = {beginWord}
    found = False
    while layer and not found:
        word_set -= layer
        next_layer = set()
        for w in layer:
            for i in range(len(w)):
                for c in 'abcdefghijklmnopqrstuvwxyz':
                    next_w = w[:i] + c + w[i+1:]
                    if next_w in word_set:
                        adj[w].append(next_w)
                        next_layer.add(next_w)
                        if next_w == endWord:
                            found = True
        layer = next_layer
    res = []
    def backtrack(curr, path):
        if curr == endWord:
            res.append(list(path))
            return
        for nxt in adj[curr]:
            path.append(nxt)
            backtrack(nxt, path)
            path.pop()
    backtrack(beginWord, [beginWord])
    return sorted(res)
```

Use BFS layer-by-layer to build a shortest-path DAG, pruning words only after completing each full level, followed by DFS to enumerate all minimal paths.
