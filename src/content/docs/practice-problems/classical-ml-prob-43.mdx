```python
def lda_scatter_matrices(class0, class1):
    dim = len(class0[0])
    m0 = [sum(p[j] for p in class0) / len(class0) for j in range(dim)]
    m1 = [sum(p[j] for p in class1) / len(class1) for j in range(dim)]
    sw = [[0.0] * dim for _ in range(dim)]
    for p in class0:
        d = [p[j] - m0[j] for j in range(dim)]
        for r in range(dim):
            for c in range(dim):
                sw[r][c] += d[r] * d[c]
    for p in class1:
        d = [p[j] - m1[j] for j in range(dim)]
        for r in range(dim):
            for c in range(dim):
                sw[r][c] += d[r] * d[c]
    d_mean = [m1[j] - m0[j] for j in range(dim)]
    sb = [[round(float(d_mean[r] * d_mean[c]), 4) for c in range(dim)] for r in range(dim)]
    sw_rounded = [[round(float(sw[r][c]), 4) for c in range(dim)] for r in range(dim)]
    return {"sw": sw_rounded, "sb": sb}
```

Fishers LDA finds projection directions that simultaneously maximize between-class variance and minimize within-class variance via the Rayleigh quotient.
