```python
def min_max_scaler(X, min_val=0.0, max_val=1.0):
    if not X or not X[0]:
        return []
    n = len(X)
    d = len(X[0])
    scaled = [[0.0] * d for _ in range(n)]
    for j in range(d):
        col = [X[i][j] for i in range(n)]
        c_min = min(col)
        c_max = max(col)
        span = c_max - c_min
        if span < 1e-12:
            for i in range(n):
                scaled[i][j] = float(min_val)
        else:
            for i in range(n):
                norm = (X[i][j] - c_min) / span
                scaled[i][j] = round(min_val + norm * (max_val - min_val), 6)
    return scaled
```

Min-Max normalization preserves zero entries if scaled to [0, 1] while bounding values strictly without assuming a normal distribution.
