```python
import math

def compute_population_stability_index(ref_counts, prod_counts):
    tot_r = sum(ref_counts)
    tot_p = sum(prod_counts)
    eps = 1e-6
    psi = 0.0
    for rc, pc in zip(ref_counts, prod_counts):
        r = max(eps, rc / tot_r) if tot_r else eps
        p = max(eps, pc / tot_p) if tot_p else eps
        psi += (p - r) * math.log(p / r)
    return {'psi': round(float(psi), 4), 'drift_detected': psi >= 0.25}
```

Normalizes frequencies and sums symmetric relative entropy across empirical bins.
