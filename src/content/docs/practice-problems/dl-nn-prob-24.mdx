```python
import math

def layer_norm_backward(dout: list[list[float]], x: list[list[float]], gamma: list[float], eps: float = 1e-5) -> dict:
    N = len(x)
    D = len(x[0])

    dx = []
    dgamma = [0.0] * D
    dbeta = [0.0] * D

    for i in range(N):
        row = x[i]
        mean = sum(row) / D
        var = sum((v - mean) ** 2 for v in row) / D
        std_inv = 1.0 / math.sqrt(var + eps)
        x_hat = [(v - mean) * std_inv for v in row]

        for j in range(D):
            dgamma[j] += dout[i][j] * x_hat[j]
            dbeta[j] += dout[i][j]

        dxhat = [dout[i][j] * gamma[j] for j in range(D)]
        s1 = sum(dxhat)
        s2 = sum(dxhat[j] * x_hat[j] for j in range(D))
        dx_row = []
        for j in range(D):
            val = (std_inv / D) * (D * dxhat[j] - s1 - x_hat[j] * s2)
            dx_row.append(round(float(val), 4))
        dx.append(dx_row)

    dgamma = [round(float(v), 4) for v in dgamma]
    dbeta = [round(float(v), 4) for v in dbeta]
    return {"dx": dx, "dgamma": dgamma, "dbeta": dbeta}
```

Differentiating through both the mean and variance terms yields the standardized projection formula for dx.
