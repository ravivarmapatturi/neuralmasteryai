```python
def best_binary_split(X_col, y):
    def gini(labels):
        if not labels:
            return 0.0
        n = len(labels)
        p1 = sum(labels) / n
        return 1.0 - p1 ** 2 - (1.0 - p1) ** 2

    n = len(y)
    base_gini = gini(y)
    pairs = sorted(zip(X_col, y), key=lambda x: x[0])
    unique_vals = sorted(list(set(X_col)))
    if len(unique_vals) <= 1:
        return 0.0, 0.0

    best_gain = -1.0
    best_thresh = unique_vals[0]
    for i in range(len(unique_vals) - 1):
        thresh = (unique_vals[i] + unique_vals[i + 1]) / 2.0
        left = [lbl for val, lbl in pairs if val <= thresh]
        right = [lbl for val, lbl in pairs if val > thresh]
        weighted_gini = (len(left) / n) * gini(left) + (len(right) / n) * gini(right)
        gain = base_gini - weighted_gini
        if gain > best_gain:
            best_gain = gain
            best_thresh = thresh
    return round(best_thresh, 4), round(best_gain, 4)
```

Scan through all midpoints between distinct sorted feature values, partition labels into left and right subsets, and select the threshold that minimizes weighted child Gini impurity.
