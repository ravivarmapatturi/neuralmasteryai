```python
def zigzag_level_order(root):
    if not root:
        return []
    result = []
    queue = [root]
    left_to_right = True
    while queue:
        vals = [n['val'] for n in queue]
        if not left_to_right:
            vals.reverse()
        result.append(vals)
        left_to_right = not left_to_right
        next_queue = []
        for n in queue:
            if n.get('left'):
                next_queue.append(n['left'])
            if n.get('right'):
                next_queue.append(n['right'])
        queue = next_queue
    return result
```

Collect nodes level by level using BFS. Use a boolean toggle to reverse the order of node values collected on odd-depth levels.
