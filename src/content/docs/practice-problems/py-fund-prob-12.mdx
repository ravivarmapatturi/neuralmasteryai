```python
def group_by_predicate(items, predicate_name):
    groups = {'true': [], 'false': []}
    for item in items:
        val = False
        if predicate_name == 'is_even':
            val = (isinstance(item, int) and item % 2 == 0)
        elif predicate_name == 'is_positive':
            val = (isinstance(item, (int, float)) and item > 0)
        elif predicate_name == 'is_nonempty_str':
            val = (isinstance(item, str) and len(item) > 0)
        if val:
            groups['true'].append(item)
        else:
            groups['false'].append(item)
    return groups
```

Type-safe conditional dispatch routes items into distinct truth partitions in a single O(n) pass.
