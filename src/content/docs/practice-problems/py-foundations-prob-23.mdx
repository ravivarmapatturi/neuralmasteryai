```python
from collections import OrderedDict

def lru_cache_access(operations, capacity):
    cache = OrderedDict()
    out = []
    for op in operations:
        name = op.get("op")
        key = op.get("key")
        if name == "get":
            if key in cache:
                cache.move_to_end(key)
                out.append(cache[key])
            else:
                out.append(-1)
        elif name == "put":
            val = op.get("val")
            if key in cache:
                cache[key] = val
                cache.move_to_end(key)
            else:
                if len(cache) >= capacity:
                    cache.popitem(last=False)
                cache[key] = val
    return out
```

OrderedDict maintains O(1) key lookups alongside O(1) moves to head/tail for maintaining LRU order.
