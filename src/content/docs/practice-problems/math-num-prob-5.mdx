```python
import math

def gram_schmidt(vectors):
    basis = []
    for v in vectors:
        w = list(v)
        for u in basis:
            dot_wu = sum(a * b for a, b in zip(w, u))
            dot_uu = sum(b * b for b in u)
            if dot_uu > 1e-9:
                scalar = dot_wu / dot_uu
                w = [a - scalar * b for a, b in zip(w, u)]
        norm_w = math.sqrt(sum(a ** 2 for a in w))
        if norm_w > 1e-7:
            u_norm = [a / norm_w for a in w]
            basis.append(u_norm)
    return [[round(float(x), 4) for x in u] for u in basis]
```

Iteratively subtracts projections against all existing orthonormal basis vectors, then normalizes.
