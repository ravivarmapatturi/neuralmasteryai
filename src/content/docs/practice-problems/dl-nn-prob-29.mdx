```python
import math

def cross_entropy_label_smoothing(logits: list[list[float]], target_idx: list[int], eps: float = 0.1) -> float:
    N = len(logits)
    C = len(logits[0])
    total_loss = 0.0

    for i in range(N):
        row = logits[i]
        m = max(row)
        exps = [math.exp(v - m) for v in row]
        sum_exp = sum(exps)
        log_probs = [v - m - math.log(sum_exp) for v in row]

        y_i = target_idx[i]
        loss_i = 0.0
        for j in range(C):
            q_j = (1.0 - eps) if j == y_i else 0.0
            q_j += eps / C
            loss_i += -q_j * log_probs[j]
        total_loss += loss_i

    return round(total_loss / N, 4)
```

Label smoothing replaces hard Dirac-delta targets with a mixture of true labels and a uniform prior over all classes.
