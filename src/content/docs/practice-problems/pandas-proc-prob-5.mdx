```python
import math

def clip_rolling_bollinger_outliers(series, window, num_std):
    clipped = []
    for i in range(len(series)):
        sub = series[max(0, i - window + 1):i + 1]
        m = sum(sub) / len(sub)
        std = math.sqrt(sum((x - m)**2 for x in sub) / (len(sub) - 1)) if len(sub) >= 2 else 0.0
        val = max(m - num_std * std, min(m + num_std * std, series[i]))
        clipped.append(round(float(val), 4))
    return clipped
```

Calculates rolling mean and standard deviation per index and clamps the current point.
