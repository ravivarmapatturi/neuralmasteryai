```python
def diff_dicts(dict_a, dict_b):
    added = {}
    removed = {}
    modified = {}
    all_keys = set(dict_a.keys()) | set(dict_b.keys())
    for k in sorted(all_keys):
        in_a = k in dict_a
        in_b = k in dict_b
        if in_b and not in_a:
            added[k] = dict_b[k]
        elif in_a and not in_b:
            removed[k] = dict_a[k]
        elif dict_a[k] != dict_b[k]:
            modified[k] = {'old': dict_a[k], 'new': dict_b[k]}
    return {'added': added, 'removed': removed, 'modified': modified}
```

Set operations on dictionary keys yield instantaneous partitioning into symmetric differences and intersections.
